% ---------- placeholder for a restricted image --------------------------
% Stands in for an image whose license does not allow sharing yet, so the
% real image never reaches any output. Draw it at the real image's printed size.
% #1 = printed width, #2 = printed height (with units),
% #3 = description of the image (plain text; write _ as \_)
% The text is 4cm wide; in a narrower box it is scaled down to fit.
\newcommand{\restrictedimage}[3]{%
  \begin{tikzpicture}
    \draw[gray, dashed, fill=gray!10] (0,0) rectangle (#1, #2);
    \node[text width=4cm, align=flush center, scale={min(1, 0.9*#1/4cm)}, transform shape]
      at (0.5*#1, 0.5*#2) {\textbf{Image pending license}\\ #3};
  \end{tikzpicture}}
